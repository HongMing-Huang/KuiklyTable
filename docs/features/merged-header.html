合并表头

GroupedHeaderTable 在标准表格上方渲染一层分组表头。每个分组单元格根据 HeaderGroup.childColumnKeys 计算 flex 总和，实现跨列合并效果。 适合财务报表、成绩单等需要多级表头的场景。

基础用法

定义列和分组信息，分组表头自动按 childColumnKeys 跨列合并：

data class ScoreReport(
    val name: String, val age: Int,
    val chinese: Int, val math: Int, val english: Int
)

GroupedHeaderTable<ScoreReport> {
    attr {
        columns = listOf(
            column("姓名", flex = 2f, key = "姓名") { it.name },
            column("年龄", flex = 1f, key = "年龄") { it.age.toString() },
            column("语文", flex = 1f, key = "语文") { it.chinese.toString() },
            column("数学", flex = 1f, key = "数学") { it.math.toString() },
            column("英语", flex = 1f, key = "英语") { it.english.toString() },
        )
        data = reports
        headerGroups = listOf(
            HeaderGroup("基本信息", listOf("姓名", "年龄")),
            HeaderGroup("考试成绩", listOf("语文", "数学", "英语")),
        )
    }
}

自定义分组表头样式

通过样式属性自定义分组表头的背景色、文字颜色等：

GroupedHeaderTable<ScoreReport> {
    attr {
        columns = scoreColumns
        data = reports
        headerGroups = listOf(
            HeaderGroup("基本信息", listOf("姓名", "年龄")),
            HeaderGroup("成绩", listOf("语文", "数学", "英语")),
        )
        groupHeaderHeight = 40f
        groupHeaderBackgroundColor = Color(0xFFE8EAF6)
        groupHeaderTextColor = Color(0xFF1A237E)
        groupHeaderFontSize = 15f
        tableInit = {
            attr {
                theme = TableTheme.BORDERED
            }
        }
    }
}

多列分组

支持任意数量的分组，每个分组可包含不同数量的列：

data class Employee(
    val id: Int, val name: String, val department: String,
    val baseSalary: Int, val bonus: Int, val total: Int
)

GroupedHeaderTable<Employee> {
    attr {
        columns = listOf(
            column("ID", flex = 1f, key = "id") { it.id.toString() },
            column("姓名", flex = 2f, key = "name") { it.name },
            column("部门", flex = 2f, key = "dept") { it.department },
            column("基本工资", flex = 1f, key = "base") { it.baseSalary.toString() },
            column("奖金", flex = 1f, key = "bonus") { it.bonus.toString() },
            column("合计", flex = 1f, key = "total") { it.total.toString() },
        )
        data = employees
        headerGroups = listOf(
            HeaderGroup("个人信息", listOf("id", "name", "dept")),
            HeaderGroup("薪资明细", listOf("base", "bonus", "total")),
        )
    }
}
